drop database if exists EscuelaDB;
create database EscuelaDB;
Use EscuelaDB;

create table Alumnos (
	alumno_id int primary key,
    nombre varchar(50) not null,
    apellido varchar(50) not null,
    fecha_nacimiento date,
    telefono varchar(15),
    email varchar(100) unique
);

create table Colegiaturas (
	colegiatura_id int primary key,
    alumno_id int,
    monto decimal(10, 2) not null,
    mes_pagado varchar(20) not null,
    fecha_pago date,
    estado_pago varchar(15) default 'Pagado',
    foreign key (alumno_id) references Alumnos(alumno_id) on delete cascade
);

create table Calificaciones (
	calificacion_id int primary key,
    alumno_id int,
    materia varchar(50) not null,
    nota decimal(4, 2) not null,
    periodo varchar(20),
    foreign key(alumno_id) references Alumnos(alumno_id) on delete cascade
);
